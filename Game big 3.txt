  function TB(x0,y0){
    var pts=[[x0,y0]],cx=x0,cy=y0;
    var api={
      flat:function(len){cx+=len;pts.push([cx,cy]);return api},
      slope:function(len,dy){var n=Math.max(2,Math.ceil(Math.abs(len)/3));for(var i=1;i<=n;i++)pts.push([cx+len*i/n,cy+dy*i/n]);cx+=len;cy+=dy;return api},
      bump:function(len,h){var n=9;for(var i=1;i<=n;i++){var tt=i/n;pts.push([cx+len*tt,cy+Math.sin(tt*Math.PI)*h])}cx+=len;pts.push([cx,cy]);return api},
      gap:function(len,depth){depth=depth||30;pts.push([cx+0.06,cy-depth]);pts.push([cx+len-0.06,cy-depth]);cx+=len;pts.push([cx,cy]);return api},
      steps:function(n,w,h){for(var i=0;i<n;i++){cx+=w;cy+=h;pts.push([cx,cy])}return api},
      x:function(){return cx},y:function(){return cy},
      done:function(){return pts}
    };
    return api;
  }

  function buildGroundMesh(pts,theme,W){
    var T=THEMES[theme]||THEMES.factory;
    var g=new THREE.Group();
    var topMat=mat(T.ground,{roughness:0.92,metalness:0.05});
    var verts=[];
    function V(x,y,z){return new THREE.Vector3(x,y,z)}
    function push(a,b,c,d){verts.push(a.x,a.y,a.z,b.x,b.y,b.z,c.x,c.y,c.z);verts.push(a.x,a.y,a.z,c.x,c.y,c.z,d.x,d.y,d.z)}
    for(var i=0;i<pts.length-1;i++){
      var A=pts[i],B=pts[i+1];
      var len=Math.hypot(B[0]-A[0],B[1]-A[1]);
      if(len<0.001)continue;
      push(V(A[0],A[1],-W),V(A[0],A[1],W),V(B[0],B[1],W),V(B[0],B[1],-W));
      var depth=3.0;
      push(V(A[0],A[1],-W),V(B[0],B[1],-W),V(B[0],B[1]-depth,-W),V(A[0],A[1]-depth,-W));
      push(V(A[0],A[1],W),V(A[0],A[1]-depth,W),V(B[0],B[1]-depth,W),V(B[0],B[1],W));
    }
    var geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.Float32BufferAttribute(verts,3));
    geo.computeVertexNormals();
    var m=new THREE.Mesh(geo,topMat);
    m.receiveShadow=true;
    g.add(m);
    return g;
  }

  function solveIK(limb,tx,ty,l1,l2){
    var d=Math.hypot(tx,ty);
    d=clamp(d,Math.abs(l1-l2)+0.02,l1+l2-0.01);
    var alpha=Math.atan2(ty,tx);
    var cosB=(d*d+l1*l1-l2*l2)/(2*d*l1);
    var B=Math.acos(clamp(cosB,-1,1));
    var phiThigh=alpha+B;
    var cosG=(l1*l1+l2*l2-d*d)/(2*l1*l2);
    var G=Math.acos(clamp(cosG,-1,1));
    var phiShin=phiThigh-(Math.PI-G);
    limb.hipPivot.rotation.z=phiThigh+Math.PI/2;
    limb.kneePivot.rotation.z=phiShin-phiThigh;
  }

  var P={gravity:22,mass:130,I:230,wheelBase:2.0,wheelR:0.5,wheelY:-0.5,suspK:15000,suspD:1500,maxDrive:1500,maxBrake:2600,maxSpeed:28,airTorque:11,groundTorque:26,drag:0.32,rollRes:14};
  var bike={x:0,y:0,vx:0,vy:0,ang:0,angVel:0,onGround:false,groundCount:0,wheelSpin:0,crank:0,airTime:0,boost:0,crashTimer:0};

  function buildWorldState(lv){
    activeObstacles=lv.obstacles||[];
    platformStates=activeObstacles.map(function(o,i){
      var st={o:o,i:i,time:Math.random()*6,y:0,x:0};

        if(o.ground){st.baseY=(groundAt(lv,o.x)||0)+(o.h/2)+0.2}

    });

  function terrainY(pts,x){
    var best=-Infinity;
    for(var i=0;i<pts.length-1;i++){
      var a=pts[i],b=pts[i+1];
      if(x>=a[0]&&x<=b[0]){
        var tt=(x-a[0])/((b[0]-a[0])||1);
        var y=a[1]+(b[1]-a[1])*tt;
        if(y>best)best=y;
      }
    }
    return best;
  }
  function groundAt(lv,x){return terrainY(lv.pts,x)}

  function surfaceBelow(x,yMax){
    var bestY=-Infinity,norm={x:0,y:1};
    var pts=level.pts;
    for(var i=0;i<pts.length-1;i++){
      var a=pts[i],b=pts[i+1];
      if(x>=a[0]&&x<=b[0]){
        var tt=(x-a[0])/((b[0]-a[0])||1);
        var y=a[1]+(b[1]-a[1])*tt;
        if(y<=yMax+0.35&&y>bestY){
          bestY=y;
          var dx=b[0]-a[0],dy=b[1]-a[1],L=Math.hypot(dx,dy)||1;
          norm={x:-dy/L,y:dx/L};
        }
      }
    }
    for(var k=0;k<platformStates.length;k++){
      var st=platformStates[k],o=st.o;
      if(o.t!=='mover')continue;
      var top=st.y+(o.h||0.7)/2;
      if(x>=st.x-(o.w||3.5)/2&&x<=st.x+(o.w||3.5)/2){
        if(top<=yMax+0.35&&top>bestY){bestY=top;norm={x:0,y:1}}
      }
    }
    return{y:bestY,norm:norm};
  }

  function frictionAt(x){
    if(level.zones){for(var i=0;i<level.zones.length;i++){var z=level.zones[i];if(x>=z.x0&&x<=z.x1)return z.mu}}
    return 1.0;
  }

  function resetBikeAt(x,y){bike.x=x;bike.y=y;bike.vx=0;bike.vy=0;bike.ang=0;bike.angVel=0;bike.wheelSpin=0;bike.airTime=0;bike.crashTimer=0;bike.boost=0}

  var FIXED=1/120;
  function stepPhysics(dt){
    var m=P.mass,I=P.I;
    var fx=0,fy=0,torque=0;
    fy-=m*P.gravity;
    fx-=bike.vx*Math.abs(bike.vx)*P.drag;
    fy-=bike.vy*Math.abs(bike.vy)*P.drag*0.4;
    var cosA=Math.cos(bike.ang),sinA=Math.sin(bike.ang);
    var wheels=[{lx:-P.wheelBase/2,ly:P.wheelY,drive:true},{lx:P.wheelBase/2,ly:P.wheelY,drive:false}];
    var anyGround=false;
    var input=Game.input;
    var throttle=input.gas*SaveManager.data.settings.ctlSens;
    var braking=input.brake;
    var lean=input.lean;
    for(var wi=0;wi<2;wi++){
      var w=wheels[wi];
      var wx=bike.x+w.lx*cosA-w.ly*sinA;
      var wy=bike.y+w.lx*sinA+w.ly*cosA;
      var surf=surfaceBelow(wx,wy+P.wheelR*0.2);
      if(surf.y===-Infinity)continue;
      var wheelBottom=wy-P.wheelR;
      var comp=surf.y-wheelBottom;
      if(comp<=0)continue;
      anyGround=true;
      var n=surf.norm;
      var rx=wx-bike.x,ry=wy-bike.y;
      var vwx=bike.vx-bike.angVel*ry;
      var vwy=bike.vy+bike.angVel*rx;
      var vn=vwx*n.x+vwy*n.y;
      var Fn=P.suspK*Math.min(comp,0.7)-P.suspD*vn;
      Fn=Math.max(0,Fn);
      fx+=Fn*n.x;fy+=Fn*n.y;
      torque+=rx*(Fn*n.y)-ry*(Fn*n.x);
      var tx=n.y,ty=-n.x;
      var vt=vwx*tx+vwy*ty;
      var mu=frictionAt(wx);
      var Ft=0;
      if(w.drive&&throttle>0){var sf=Math.max(0,1-Math.abs(vt)/P.maxSpeed);Ft+=throttle*P.maxDrive*sf}
      if(w.drive&&throttle<0)Ft+=throttle*P.maxDrive*0.45;
      if(braking>0){var bs=Math.sign(vt);if(Math.abs(vt)>0.3)Ft-=bs*braking*P.maxBrake;else Ft-=vt*400}
      Ft-=vt*P.rollRes;
      var maxF=mu*Fn*1.35+40;
      Ft=clamp(Ft,-maxF,maxF);
      fx+=Ft*tx;fy+=Ft*ty;
      torque+=rx*(Ft*ty)-ry*(Ft*tx);
      bike.wheelSpin+=vt*dt/P.wheelR;
    }
    if(!anyGround){torque+=lean*P.airTorque*P.mass*0.06;bike.airTime+=dt}
    else{bike.airTime=0;torque+=lean*P.groundTorque*P.mass*0.035}
    torque-=bike.angVel*260;
    bike.vx+=fx/m*dt;
    bike.vy+=fy/m*dt;
    bike.angVel+=torque/I*dt;
    bike.angVel=clamp(bike.angVel,-14,14);
    bike.x+=bike.vx*dt;
    bike.y+=bike.vy*dt;
    bike.ang+=bike.angVel*dt;
    bike.onGround=anyGround;
    if(anyGround)bike.groundCount++;else bike.groundCount=0;
  }

  function checkCollisions(){
    if(bike.crashTimer>0)return;
    var cosA=Math.cos(bike.ang),sinA=Math.sin(bike.ang);
    var pts=[
      {x:bike.x-0.9*cosA+0.5*sinA,y:bike.y-0.9*sinA-0.5*cosA,r:0.42},
      {x:bike.x+0.9*cosA+0.5*sinA,y:bike.y+0.9*sinA-0.5*cosA,r:0.42},
      {x:bike.x,y:bike.y,r:0.5}
    ];
    for(var i=0;i<platformStates.length;i++){
      var st=platformStates[i],o=st.o;
      if(o.t==='mover'||o.t==='fan'||o.t==='boost'||o.t==='spring')continue;
      if(o.t==='saw'){
        var cx=o.x,cy=(o.ground?groundAt(level,o.x)+(o.r||1.2):cy);
        for(var p=0;p<pts.length;p++){var pp=pts[p];var d=Math.hypot(pp.x-cx,pp.y-cy);if(d<(o.r||1.2)*0.88+pp.r)return crash()}
        continue;
      }
      if(o.t==='hammer'){
        var ax=o.x,ay=(o.ground?groundAt(level,o.x):0)+(o.anchorH||4.2);
        var ang=st.angle;
        var hx=ax+Math.sin(ang)*(o.len||3);
        var hy=ay-Math.cos(ang)*(o.len||3);
        for(var p2=0;p2<pts.length;p2++){var pp2=pts[p2];if(Math.hypot(pp2.x-hx,pp2.y-hy)<0.95+pp2.r)return crash()}
        continue;
      }
      if(o.t==='bar'){
        var bx=o.x+(o.off||0),by=groundAt(level,o.x)+(o.h||3)/2;
        var bw=0.35,bh=o.h||3;
        for(var p3=0;p3<pts.length;p3++){var pp3=pts[p3];if(Math.abs(pp3.x-bx)<bw/2+pp3.r&&Math.abs(pp3.y-by)<bh/2+pp3.r)return crash()}
        continue;
      }
      var ccx=o.x,ccy,ww,hh;
      if(o.t==='spike'){ww=o.w||1.5;hh=o.h||0.9;ccy=(o.ground?groundAt(level,o.x):o.y||0)+hh/2}
      else{ww=o.w||1;hh=o.h||1;ccy=(o.ground?groundAt(level,o.x):(o.y||0))+hh/2}
      for(var p4=0;p4<pts.length;p4++){
        var pp4=pts[p4];
        var dx=Math.abs(pp4.x-ccx)-(ww/2);
        var dy=Math.abs(pp4.y-ccy)-(hh/2);
        if(dx<pp4.r&&dy<pp4.r){
          var ddx=Math.max(dx,0),ddy=Math.max(dy,0);
          if(ddx*ddx+ddy*ddy<pp4.r*pp4.r)return crash();
        }
      }
    }
  }

  var particlePool=[];
  var particleGeo=new THREE.SphereGeometry(0.09,5,4);
  function spawnParticles(x,y,color,count,spread){
    if(quality==='low')count=Math.ceil(count*0.4);
    for(var i=0;i<count;i++){
      var p=null;
      for(var k=0;k<particlePool.length;k++)if(!particlePool[k].alive){p=particlePool[k];break}
      if(!p){
        if(particlePool.length>140)break;
        var m=new THREE.Mesh(particleGeo,new THREE.MeshBasicMaterial({color:color,transparent:true}));
        worldRoot.add(m);
        p={mesh:m,alive:false};particlePool.push(p);
      }
      p.alive=true;p.life=0.55+Math.random()*0.4;p.maxLife=p.life;
      p.mesh.visible=true;
      p.mesh.position.set(x+(Math.random()-0.5)*0.5,y+(Math.random()-0.5)*0.5,(Math.random()-0.5)*0.8);
      p.vx=(Math.random()-0.5)*spread;
      p.vy=(Math.random()*0.7+0.3)*spread*0.6;
      p.mesh.scale.setScalar(0.6+Math.random()*0.9);
    }
  }
  function updateParticles(dt){
    for(var i=0;i<particlePool.length;i++){
      var p=particlePool[i];
      if(!p.alive)continue;
      p.life-=dt;
      if(p.life<=0){p.alive=false;p.mesh.visible=false;continue}
      p.vy-=18*dt;
      p.mesh.position.x+=p.vx*dt;
      p.mesh.position.y+=p.vy*dt;
      var a=p.life/p.maxLife;
      p.mesh.material.opacity=a;
      p.mesh.scale.setScalar(0.3+a*0.9);
    }
  }

      var lv={id:'custom_test',cat:'custom',idx:1,theme:'factory',diff:2,name:{fa:'تست',en:'Test'},pts:this.groundPts,finish:this.finishX,obstacles:this.objects,cps:[],targets:[999,999,999]};
      this.active=false;removeEditorListeners();UI.hideHud();Game.start(lv);
    }
  };

  var editorListeners=[];
  function setupEditorListeners(){
    var onPointer=function(e){
      if(!Editor.active)return;
      var cx=(e.changedTouches?e.changedTouches[0].clientX:e.clientX);
      var cy=(e.changedTouches?e.changedTouches[0].clientY:e.clientY);
      var ndc=new THREE.Vector2((cx/innerWidth)*2-1,-(cy/innerHeight)*2+1);
      var ray=new THREE.Raycaster();
      ray.setFromCamera(ndc,camera);
      var plane=new THREE.Plane(new THREE.Vector3(0,0,1),0);
      var pt=new THREE.Vector3();
      if(ray.ray.intersectPlane(plane,pt)){
        var hits=ray.intersectObjects(Editor.visuals,true);
        if(hits.length){
          var obj=hits[0].object;
          while(obj.parent&&Editor.visuals.indexOf(obj)<0)obj=obj.parent;
          var idx=Editor.visuals.indexOf(obj);
          if(idx>=0){Editor.selected=idx;Editor.renderObjects();return}
        }
        Editor.addObject(pt.x,pt.y);
      }
    };
    var onKey=function(e){
      if(!Editor.active)return;
      if(e.key==='Delete'||e.key==='Backspace')Editor.deleteSelected();
      if(e.key==='z'&&(e.ctrlKey||e.metaKey)){e.preventDefault();Editor.undo()}
      if(e.key==='y'&&(e.ctrlKey||e.metaKey)){e.preventDefault();Editor.redo()}
    };
    canvas.addEventListener('pointerdown',onPointer);
    addEventListener('keydown',onKey);
    editorListeners=[['pointerdown',onPointer],['keydown',onKey]];
  }
  function removeEditorListeners(){
    if(editorListeners.length){
      canvas.removeEventListener(editorListeners[0][1].__ev||'pointerdown',editorListeners[0][1]);
      removeEventListener('keydown',editorListeners[1][1]);
    }
  }

  // ============ MAIN LOOP ============
  var lastT=performance.now();
  var frames=0,fpsAcc=0;
  function loop(now){
    requestAnimationFrame(loop);
    var dt=Math.min(0.05,(now-lastT)/1000);
    lastT=now;
    Game.frameCount++;
    if(Game.state==='playing'){
      if(isTouch){Game.input.gas=touchState.gas;Game.input.brake=touchState.brake;Game.input.lean=touchState.lean}
      else{var k=readKeyboard();Game.input.gas=k.gas;Game.input.brake=k.brake;Game.input.lean=k.lean}
      Game.update(dt);
      AudioMan.engineUpdate(Math.hypot(bike.vx,bike.vy),Game.input.gas,bike.onGround);
      frames++;fpsAcc+=dt;
      if(fpsAcc>1){var f=frames/fpsAcc;frames=0;fpsAcc=0;if(f<32&&quality==='high'){quality='medium';applyQuality();SaveManager.data.settings.quality='medium';SaveManager.save()}}
    } else if(Editor.active){
      var t2=now*0.00008;
      camera.position.x=Editor.groundPts[0][0]+(Editor.groundPts[Editor.groundPts.length-1][0]-Editor.groundPts[0][0])*0.5+Math.sin(t2)*10;
      camera.lookAt(camera.position.x,4,0);
    }
    renderer.render(scene,camera);
  }
